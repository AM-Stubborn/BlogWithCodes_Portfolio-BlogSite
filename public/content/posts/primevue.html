Data table

Installation

npm install primevue@^3 --save npm install primeicons --save

Configuration

In main.js

import PrimeVue from 'primevue/config';

import DataTable from 'primevue/datatable';

import Column from 'primevue/column';

import Paginator from 'primevue/paginator';

const app = createApp(App); app.use(PrimeVue);

app.component('DataTable', DataTable); app.component('Paginator', Paginator); app.component('Column', Column);


Styling

In App.vue

import "primevue/resources/themes/saga-blue/theme.css";
import "primevue/resources/primevue.min.css";
import "primeicons/primeicons.css";

Example

<template>

<div class="mx-4 ">

<DataTable :value="users"

:paginator="false"

:rows="pageSize"

:totalRecords="totalRecords"

class="p-datatable-gridlines"

:scrollable="true"

scrollHeight="700px"

:lazy="true"

:rowsPerPageOptions="pageSizeList"

>

<Column field="name" header="name" style="min-width:260px" :sortable="true"> </Column>

<Column field="email" header="email" style="min-width: 260px" :sortable="true"></Column>

<Column field="address" header="address" style="min-width: 260px"></Column>

<Column field="address2" header="address2" style="min-width: 260px"></Column>

<Column field="city" header="city" style="min-width: 260px" :sortable="true"></Column>

<Column field="zip" header="zip" style="min-width: 260px" :sortable="true"></Column>

<Column header="Action" alignFrozen="right" style="min-width: 150px; display: flex; justify-content: space-between;" :frozen="true">

<template #body="{data}">

<i class="fa-sharp fa-solid fa-trash"></i>

<i class="fa-solid fa-check"></i>

<i class="fa-solid fa-key"></i>

</template>

</Column>

<template #footer>

<Paginator :template="{

default: 'FirstPageLink PrevPageLink PageLinks NextPageLink LastPageLink RowsPerPageDropdown'

}"

:rows="pageSize"

:totalRecords="totalRecords"

:rowsPerPageOptions="pageSizeList"

@page="onPage($event)"

@sort="onSort($event)"

>

<template #start>

Total Records: {{totalRecords}}

</template>

</Paginator>

</template>

</DataTable>

</div>

</template>


<script>


import dbStorage from "../../services/DataBase.js"


export default {

name: 'PrimeDataTable',

data() {

return {

users: null,

totalRecords: 0,

pageNumber: 0,

pageSize: 20,

pageSizeList:[10,20,30]

}

},

created() {

this.totalRecords = dbStorage.getTotalCount();

this.users = dbStorage.GetUsers(this.pageSize, this.pageNumber + 1,'',1);


},

mounted() {

},

methods: {

onPage(event) {

console.log('page', event)

this.users = dbStorage.GetUsers(event.rows, event.page + 1,'',1);

},

onSort(event) {

console.log('Sort', event)

this.users = dbStorage.GetUsers(this.pageSize, this.pageNumber + 1, event.sortField, event.sortOrder);


}

}

}


</script>
